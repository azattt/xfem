! ============================================================
! SMART / CINT 3D EDGE CRACK MODEL
!
! Full plate:
! X: 0 ... W
! Y: 0 ... H
! Z: 0 ... T
!
! Edge crack:
! x = 0 ... a
! y = H/2
! z = 0 ... T
!
! Loading:
! top face    y = H : tension +Y
! bottom face y = 0 : tension -Y
!
! Element:
! SOLID187 tetrahedral mesh
! ============================================================

! -----------------------------
! Parameters

*SET,W,1.0
*SET,H,1.0
*SET,T,0.1

*SET,a,0.3
*SET,yc,H/2

*SET,sigma,1000000

! Coarse value 0.1 is too rough for crack front.
! Use 0.03 first. Then refine to 0.02 if needed.
*SET,mesh_size,0.03
*SET,crack_front_size,0.01

! For test growth.
! If crack does not grow: decrease KIC or increase sigma.
! If solution becomes unstable: increase KIC or decrease sigma.
*SET,KIC,1.0e5

! -----------------------------
! Element and material
! -----------------------------
ET,1,SOLID187

MP,EX,1,2e11
MP,PRXY,1,0.3

TYPE,1
MAT,1

! GEOMETRY
!
! Four volumes:
! V_LL: x=0..a, y=0..yc
! V_LU: x=0..a, y=yc..H
! V_RL: x=a..W, y=0..yc
! V_RU: x=a..W, y=yc..H
!
! Crack faces:
! between V_LL and V_LU for x=0..a, y=yc.
!
! These crack-face nodes must remain duplicated.

BLOCK,0,a,0,yc,0,T
*GET,v_ll,VOLU,0,NUM,MAX

BLOCK,0,a,yc,H,0,T
*GET,v_lu,VOLU,0,NUM,MAX

BLOCK,a,W,0,yc,0,T
*GET,v_rl,VOLU,0,NUM,MAX

BLOCK,a,W,yc,H,0,T
*GET,v_ru,VOLU,0,NUM,MAX

ALLSEL,ALL

! We merge only material-continuity interfaces:
!
! 1) x = a, y = 0..yc
! 2) x = a, y = yc..H
! 3) y = yc, x = a..W

! We do NOT merge crack faces:
! y = yc, x = 0..a

ALLSEL,ALL

NSEL,R,LOC,Y,yc,H
NUMMRG,NODE
ALLSEL,ALL

! ------------------------------------------------------------
! Merge right ligament: y = yc, x = a..W
! ------------------------------------------------------------
NSEL,S,LOC,Y,yc
NSEL,R,LOC,X,a,W
NUMMRG,NODE
ALLSEL,ALL

NUMCMP,NODE
NUMCMP,ELEM

ALLSEL,ALL

! ============================================================
! LOADS
! ============================================================

ASEL,S,LOC,Y,H
SFA,ALL,1,PRES,-sigma
ALLSEL,ALL

! Bottom surface: tension downward
ASEL,S,LOC,Y,0
SFA,ALL,1,PRES,-sigma
ALLSEL,ALL

D,nA,UX,0

NSEL,S,LOC,X,W

*GET,nB,NODE,0,NUM,MIN

D,nB,UY,0
D,nB,UZ,0

ALLSEL,ALL

NSEL,S,LOC,X,0
NSEL,R,LOC,Y,0
NSEL,R,LOC,Z,T
*GET,nC,NODE,0,NUM,MIN

D,nC,UY,0

ALLSEL,ALL
